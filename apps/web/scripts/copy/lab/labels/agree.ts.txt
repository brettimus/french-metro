// Compares a run against both editors' grades. Scores use the final row dims (level = 1 + norm*(L-1)).
const S = "/private/tmp/claude-501/-Users-brettbeutell-fiber-french-metro/48ab2d32-bbcf-43be-9eed-2b238b868894/scratchpad";
const R = await Bun.file(process.argv[2]!).json();

const FR = await Bun.file(`${S}/fr-calib-grader/mine.json`).json();
const rows = new Map<string, any>(R.rows.map((r: any) => [r.id, r]));
const LEVELS: Record<string, number> = { S1: 4, S2: 4, S3: 4, S4: 4, S5: 3, S6: 4, S7: 4, L1: 4, L2: 4, L3: 3, U1: 3, U3: 3 };
const TELLS = ["t2_inflated","t4_praise","t5_neg_parallel","t6_tricolon","t7_vague_attribution","t9_closer","t12_false_range","t13_stock_metaphor","t14_grandiosity","t16_synonym_cycling"];
type Item = { who: "en" | "fr"; unit: string; q: string; kind: "score" | "noul"; mine: number; jev: number | undefined; raw?: number };
const items: Item[] = [];
const dimLevel = (row: any, q: string) => (row?.dims?.[q] === undefined ? undefined : 1 + row.dims[q] * (LEVELS[q]! - 1));
const rawLevel = (row: any, k: string) => { const a = row?.jev?.[k]; return a?.type === "score" ? a.score + 1 : undefined; };
const noul = (row: any, k: string) => { const a = row?.jev?.[k]; return a?.type === "noul" ? a.noul : undefined; };
const codeCalque = (row: any) => (row?.findings ?? []).some((f: any) => f.check === "fr.calquePattern" || f.check === "parity.frCalque") ? 1 : 0;

for (const [who, G] of [["en", EN], ["fr", FR]] as const) {
  for (const [id, g] of Object.entries<any>(G.units)) {
    const row = rows.get(id);
    for (const q of Object.keys(LEVELS)) if (g[q] !== undefined) items.push({ who, unit: id, q, kind: "score", mine: g[q], jev: dimLevel(row, q), raw: rawLevel(row, `unit.${q}`) });
    if (g.tells) for (const t of TELLS) if (row?.jev?.[`unit.${t}`]) items.push({ who, unit: id, q: t, kind: "noul", mine: g.tells.includes(t) ? 1 : 0, jev: noul(row, `unit.${t}`) });
    if (g.native !== undefined) items.push({ who, unit: id, q: "native", kind: "score", mine: g.native, jev: rawLevel(row, "unit.native") });
    if (g.calque !== undefined) {
      items.push({ who, unit: id, q: "t19_calque(jev)", kind: "noul", mine: g.calque, jev: noul(row, "unit.t19_calque") });
      items.push({ who, unit: id, q: "calque(code)", kind: "noul", mine: g.calque, jev: codeCalque(row) });
      if (row?.jev?.["unit.metonymy"]) items.push({ who, unit: id, q: "calque(jev t19|metonymy|code)", kind: "noul", mine: g.calque, jev: Math.max(noul(row, "unit.t19_calque") ?? 0, noul(row, "unit.metonymy") ?? 0, codeCalque(row)) });
    }
  }
  for (const [pid, g] of Object.entries<any>(G.pairs)) {
    const row = rows.get(`${pid}/en`);
    const s6 = g.S6_judged ?? g.S6;
    if (s6 !== undefined) {
      items.push({ who, unit: pid, q: "S6(derived)", kind: "score", mine: s6, jev: dimLevel(row, "S6") });
      items.push({ who, unit: pid, q: "S6_judged", kind: "score", mine: s6, jev: rawLevel(row, "pair.S6_judged") });
    }
    if (g.U3 !== undefined) items.push({ who, unit: pid, q: "U3", kind: "score", mine: g.U3, jev: dimLevel(row, "U3") });
    for (const q of ["fr_missing", "en_missing", "conflict", "hedge_mismatch", "fr_calque"]) {
      const m = who === "en" ? (g.yes ? (g.yes.includes(q) ? 1 : 0) : undefined) : g[q === "hedge_mismatch" ? "hedge" : q];
      const j = noul(row, `pair.${q}`);
      if (m !== undefined && j !== undefined) items.push({ who, unit: pid, q: `pair.${q}`, kind: "noul", mine: m, jev: j });
    }
  }
}

const flagged = new Set(F.map((x) => x.join("|")));

function stats(list: Item[]) {
  const by = new Map<string, Item[]>();
  for (const it of list) if (it.jev !== undefined) by.set(`${it.who}:${it.q}`, [...(by.get(`${it.who}:${it.q}`) ?? []), it]);
  const out: Record<string, string> = {};
  for (const [k, xs] of [...by].sort()) {
    if (xs[0]!.kind === "score") {
      const mad = xs.reduce((a, x) => a + Math.abs(x.jev! - x.mine), 0) / xs.length;
      const bias = xs.reduce((a, x) => a + (x.jev! - x.mine), 0) / xs.length;
      const exact = xs.filter((x) => Math.round(x.jev!) === x.mine).length;
      out[k] = `n=${xs.length} MAD=${mad.toFixed(2)} exact=${exact}/${xs.length} bias=${bias.toFixed(2)}`;
    } else {
      const agree = xs.filter((x) => (x.jev! >= 0.5 ? 1 : 0) === x.mine).length;
      const tp = xs.filter((x) => x.mine && x.jev! >= 0.5).length, pos = xs.filter((x) => x.mine).length;
      const fp = xs.filter((x) => !x.mine && x.jev! >= 0.5).length;
      out[k] = `n=${xs.length} agree=${agree}/${xs.length} TP=${tp}/${pos} FP=${fp}`;
    }
  }
  return out;
}
